static func _填槽(模板: String) -> String:
	var 地: String = 地名.pick_random()
	var 人: String = 人名.pick_random()
	var 妖: String = 妖名.pick_random()
	var 物: String = 宝名.pick_random()
	var s := 模板.replace("{地}", 地).replace("{人}", 人).replace("{妖}", 妖).replace("{物}", 物)
	return s

static func 取历练(等阶: String, 性格 := "") -> String:
	var 池: Array = 历练模板.get(等阶, 历练模板["练气"])
	var 模板: String = 池.pick_random()
	return _填槽(模板)
